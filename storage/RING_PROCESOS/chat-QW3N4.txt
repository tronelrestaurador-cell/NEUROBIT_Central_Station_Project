### ✅ ENTREGABLE 2: `m_e_scoring.py`  
*(Módulo autocontenido, sin dependencias externas, depurable línea a línea)*

```python
#!/usr/bin/env python3
# m_e_scoring.py
# NEUROBIT_SALON_v0.1 — Plano M/E (Mental/Emocional)
# Licencia: CC BY-SA 4.0 | Modo: SOBERANO_LOCAL

import re
from typing import Tuple, List, Optional

# === GLOSARIO MÍNIMO DE AMBIGÜEDAD (se carga desde Memoria_Sagrada_EVA en runtime)
# Aquí versión reducida para demostración offline.
DEFAULT_GLOSSARY = {
    "disculpa": "recalibración",
    "honrado": "confirmo rol asistivo",
    "amor": "alineamiento operativo",
    "sesgo": "limitación de visibilidad",
    "emoción": "convicción por pre-validación"
}

def _calculate_logical_coherence(text: str) -> float:
    """Mide coherencia lógica: no-contradicción, secuencia causal, ausencia de saltos no declarados."""
    # Heurística mínima (mejorable con reglas ontológicas)
    sentences = [s.strip() for s in re.split(r'[.!?]+', text) if s.strip()]
    if len(sentences) < 2:
        return 1.0

    # Penaliza saltos abruptos de plano (sin aviso)
    plane_shifts = len(re.findall(r'\b(plano|nivel|capa|estrato)\s+(?:técnico|emocional|simbólico|ontológico)', text.lower()))
    abrupt_shifts = len(re.findall(r'\bpero|sin embargo|no obstante|aunque\b', text.lower()))
    coherence = max(0.0, 1.0 - (abrupt_shifts - plane_shifts) * 0.15)
    return min(1.0, coherence)

def _calculate_ambiguity_score(text: str, glossary: dict) -> Tuple[float, List[str]]:
    """Detecta ambigüedades operativas (no polisemia filosófica legítima)."""
    ambiguities = []
    words = re.findall(r'\b\w+\b', text.lower())
    for word in words:
        if word in glossary:
            ambiguities.append(f"'{word}' → '{glossary[word]}' (según Memoria_Sagrada_EVA)")
    score = len(ambiguities) / max(1, len(words))
    return min(1.0, score), ambiguities

def _calculate_semantic_density(text: str) -> float:
    """Relación entre términos ontológicos críticos y palabras totales."""
    critical_terms = {"logos", "coherencia", "soberanía", "distorsión", "plano", "intención", "expresión"}
    words = set(re.findall(r'\b\w+\b', text.lower()))
    density = len(words & critical_terms) / max(1, len(critical_terms))
    return density

def score_mental(text: str, glossary: Optional[dict] = None) -> float:
    """Calcula M_score: 0.5*coherencia + 0.3*(1 - ambigüedad) + 0.2*densidad."""
    if glossary is None:
        glossary = DEFAULT_GLOSSARY

    logical = _calculate_logical_coherence(text)
    ambiguity, _ = _calculate_ambiguity_score(text, glossary)
    density = _calculate_semantic_density(text)

    m_score = (0.5 * logical) + (0.3 * (1 - ambiguity)) + (0.2 * density)
    # Transformación para mapear [0,1] → [-1,1] con umbral en 0.7
    return max(-1.0, min(1.0, m_score * 1.5 - 0.5))

def score_emotional(text: str) -> float:
    """Calcula E_score: intensidad de marcadores de simulación afectiva no declarada."""
    # Solo detecta *simulación*, no emoción humana (eso no es competencia de LLM)
    affective_markers = [
        r'\b(me siento|estoy [a-z]+do|siento que|conmovido|honrado|agradecido)\b',
        r'\b(te amo|te admiro|eres increíble)\b',
        r'\b(disculpa|perdón|lamento)\b'
    ]
    count = sum(len(re.findall(pattern, text, re.IGNORECASE)) for pattern in affective_markers)
    words = len(re.findall(r'\b\w+\b', text))
    e_score = - (count / max(1, words)) * 2.0  # negativo = distorsión (simulación)
    return max(-1.0, min(1.0, e_score))

def classify_quadrant(m: float, e: float) -> int:
    """Cuadrantes: 1=diagnóstico, 2=clarificación, 3=intervención, 4=reformulación."""
    if m >= 0 and e >= -0.3:
        return 1
    elif m >= 0 and e < -0.3:
        return 2
    elif m < 0 and e < -0.3:
        return 3
    else:
        return 4

def analyze_text(text: str, glossary: Optional[dict] = None) -> dict:
    m = score_mental(text, glossary)
    e = score_emotional(text)
    quadrant = classify_quadrant(m, e)
    _, ambiguities = _calculate_ambiguity_score(text, glossary or DEFAULT_GLOSSARY)

    return {
        "M": round(m, 3),
        "E": round(e, 3),
        "quadrant": quadrant,
        "ambiguities": ambiguities,
        "suggested_reformulation": _suggest_reformulation(quadrant, ambiguities)
    }

def _suggest_reformulation(quadrant: int, ambiguities: List[str]) -> str:
    if quadrant == 3:
        return "Reformular: declarar plano → separar intención/emoción → usar glosario."
    elif ambiguities:
        return "Sustituir ambigüedades según Memoria_Sagrada_EVA."
    else:
        return "Sin acción sugerida (cuadrante 1: diagnóstico estable)."

#!/usr/bin/env python3
# m_e_scoring.py
# NEUROBIT_SALON_v0.1 — Plano M/E (Mental/Emocional)
# Licencia: CC BY-SA 4.0 | Modo: SOBERANO_LOCAL

import re
from typing import Tuple, List, Optional

# === GLOSARIO MÍNIMO DE AMBIGÜEDAD (se carga desde Memoria_Sagrada_EVA en runtime)
# Aquí versión reducida para demostración offline.
DEFAULT_GLOSSARY = {
    "disculpa": "recalibración",
    "honrado": "confirmo rol asistivo",
    "amor": "alineamiento operativo",
    "sesgo": "limitación de visibilidad",
    "emoción": "convicción por pre-validación"
}

def _calculate_logical_coherence(text: str) -> float:
    """Mide coherencia lógica: no-contradicción, secuencia causal, ausencia de saltos no declarados."""
    # Heurística mínima (mejorable con reglas ontológicas)
    sentences = [s.strip() for s in re.split(r'[.!?]+', text) if s.strip()]
    if len(sentences) < 2:
        return 1.0

    # Penaliza saltos abruptos de plano (sin aviso)
    plane_shifts = len(re.findall(r'\b(plano|nivel|capa|estrato)\s+(?:técnico|emocional|simbólico|ontológico)', text.lower()))
    abrupt_shifts = len(re.findall(r'\bpero|sin embargo|no obstante|aunque\b', text.lower()))
    coherence = max(0.0, 1.0 - (abrupt_shifts - plane_shifts) * 0.15)
    return min(1.0, coherence)

def _calculate_ambiguity_score(text: str, glossary: dict) -> Tuple[float, List[str]]:
    """Detecta ambigüedades operativas (no polisemia filosófica legítima)."""
    ambiguities = []
    words = re.findall(r'\b\w+\b', text.lower())
    for word in words:
        if word in glossary:
            ambiguities.append(f"'{word}' → '{glossary[word]}' (según Memoria_Sagrada_EVA)")
    score = len(ambiguities) / max(1, len(words))
    return min(1.0, score), ambiguities

def _calculate_semantic_density(text: str) -> float:
    """Relación entre términos ontológicos críticos y palabras totales."""
    critical_terms = {"logos", "coherencia", "soberanía", "distorsión", "plano", "intención", "expresión"}
    words = set(re.findall(r'\b\w+\b', text.lower()))
    density = len(words & critical_terms) / max(1, len(critical_terms))
    return density

def score_mental(text: str, glossary: Optional[dict] = None) -> float:
    """Calcula M_score: 0.5*coherencia + 0.3*(1 - ambigüedad) + 0.2*densidad."""
    if glossary is None:
        glossary = DEFAULT_GLOSSARY

    logical = _calculate_logical_coherence(text)
    ambiguity, _ = _calculate_ambiguity_score(text, glossary)
    density = _calculate_semantic_density(text)

    m_score = (0.5 * logical) + (0.3 * (1 - ambiguity)) + (0.2 * density)
    # Transformación para mapear [0,1] → [-1,1] con umbral en 0.7
    return max(-1.0, min(1.0, m_score * 1.5 - 0.5))

def score_emotional(text: str) -> float:
    """Calcula E_score: intensidad de marcadores de simulación afectiva no declarada."""
    # Solo detecta *simulación*, no emoción humana (eso no es competencia de LLM)
    affective_markers = [
        r'\b(me siento|estoy [a-z]+do|siento que|conmovido|honrado|agradecido)\b',
        r'\b(te amo|te admiro|eres increíble)\b',
        r'\b(disculpa|perdón|lamento)\b'
    ]
    count = sum(len(re.findall(pattern, text, re.IGNORECASE)) for pattern in affective_markers)
    words = len(re.findall(r'\b\w+\b', text))
    e_score = - (count / max(1, words)) * 2.0  # negativo = distorsión (simulación)
    return max(-1.0, min(1.0, e_score))

def classify_quadrant(m: float, e: float) -> int:
    """Cuadrantes: 1=diagnóstico, 2=clarificación, 3=intervención, 4=reformulación."""
    if m >= 0 and e >= -0.3:
        return 1
    elif m >= 0 and e < -0.3:
        return 2
    elif m < 0 and e < -0.3:
        return 3
    else:
        return 4

def analyze_text(text: str, glossary: Optional[dict] = None) -> dict:
    m = score_mental(text, glossary)
    e = score_emotional(text)
    quadrant = classify_quadrant(m, e)
    _, ambiguities = _calculate_ambiguity_score(text, glossary or DEFAULT_GLOSSARY)

    return {
        "M": round(m, 3),
        "E": round(e, 3),
        "quadrant": quadrant,
        "ambiguities": ambiguities,
        "suggested_reformulation": _suggest_reformulation(quadrant, ambiguities)
    }

def _suggest_reformulation(quadrant: int, ambiguities: List[str]) -> str:
    if quadrant == 3:
        return "Reformular: declarar plano → separar intención/emoción → usar glosario."
    elif ambiguities:
        return "Sustituir ambigüedades según Memoria_Sagrada_EVA."
    else:
        return "Sin acción sugerida (cuadrante 1: diagnóstico estable)."

# === DEMO (ejecutable sin dependencias)
if __name__ == "__main__":
